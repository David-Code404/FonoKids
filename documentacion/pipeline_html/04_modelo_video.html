  298  def masked_mean_pool(x, src_key_padding_mask):
  299      """Global Average Pooling que IGNORA los frames de padding.
  300  
  301      x.mean(dim=1) directo diluye la representación con los frames en cero
  302      del padding -- acá se promedia solo sobre los frames reales, usando la
  303      máscara invertida (src_key_padding_mask es True en el padding).
  304      """
  305      if src_key_padding_mask is None:
  306          return x.mean(dim=1)
  307  
  308      real_mask = (~src_key_padding_mask).unsqueeze(-1).float()   # (B, T, 1), 1=real, 0=padding
  309      summed = (x * real_mask).sum(dim=1)                          # (B, hidden_dim)
  310      counts = real_mask.sum(dim=1).clamp(min=1.0)                 # (B, 1) -- evita /0 si T real=0
  311      return summed / counts

Ver código · scripts/lip_reading_dataset.py:58-71
   99      def _apply_augmentation(self, sequence):
  100          """Aumento de datos SOLO para entrenamiento -- opera sobre la
  101          secuencia (T, landmark_dim) ANTES de hacer padding, para no meter
  102          ruido en los frames de relleno."""
  103          # Ruido gaussiano leve (simula jitter de detección de landmarks).
  104          # Valor chico a propósito: la señal normalizada ya es de por sí
  105          # pequeña (std temporal típica ~0.02-0.03), así que un ruido más
  106          # fuerte que eso tapa el movimiento real en vez de robustecer.
  107          sequence = sequence + np.random.normal(0, 0.003, sequence.shape).astype(np.float32)
  108  
  109          # Shift/scale aleatorio leve de las coordenadas (variación residual
  110          # de encuadre/escala que la normalización geométrica no cubre del
  111          # todo) -- valores chicos a propósito, para no destruir la señal.
  112          shift = np.random.uniform(-0.01, 0.01, size=(1, sequence.shape[1])).astype(np.float32)
  113          scale = np.random.uniform(0.97, 1.03)
  114          sequence = sequence * scale + shift
  115  
  116          # Time masking: enmascara un 8-12% de los frames a cero, simulando
  117          # oclusiones o frames sin detección -- fuerza al modelo a no
  118          # depender de un frame puntual.
  119          T = sequence.shape[0]
  120          n_mask = max(1, int(round(T * np.random.uniform(0.08, 0.12))))
  121          n_mask = min(n_mask, T)
  122          mask_idx = np.random.choice(T, size=n_mask, replace=False)
  123          sequence[mask_idx] = 0.0
  124  
  125          return sequence.astype(np.float32)

  370  def run_epoch(model, loader, criterion, optimizer, device, train):
  371      model.train() if train else model.eval()
  372      total_loss, correct, total = 0.0, 0, 0
  373  
  374      context = torch.enable_grad() if train else torch.no_grad()
  375      with context:
  376          for sequences, labels, lengths in loader:
  377              sequences = sequences.to(device)
  378              labels = labels.to(device)
  379              lengths = lengths.to(device)
  380              mask = make_padding_mask(lengths, sequences.shape[1], device)
  381  
  382              if train:
  383                  optimizer.zero_grad()
  384  
  385              outputs = model(sequences, src_key_padding_mask=mask)
  386              loss = criterion(outputs, labels)
  387  
  388              if train:
  389                  loss.backward()
  390                  # Gradient clipping -- los Transformer/Conformer entrenados desde
  391                  # cero son propensos a picos de gradiente que hacen "rebotar" la
  392                  # pérdida en vez de converger suavemente (justo lo que se veía:
  393                  # Val Loss subiendo y bajando sin mejorar). Esto lo estabiliza.
  394                  torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
  395                  optimizer.step()
  396  
  397              total_loss += loss.item() * sequences.size(0)
  398              _, predicted = outputs.max(1)
  399              correct += predicted.eq(labels).sum().item()
  400              total += labels.size(0)
  401  
  402      avg_loss = total_loss / total if total > 0 else 0.0
  403      acc = 100.0 * correct / total if total > 0 else 0.0
  404      return avg_loss, acc

  305  def build_warmup_cosine_scheduler(optimizer, epochs, warmup_epochs=5):
  306      warmup_epochs = max(1, min(warmup_epochs, epochs - 1)) if epochs > 1 else 1
  307  
  308      def lr_lambda(epoch):
  309          if epoch < warmup_epochs:
