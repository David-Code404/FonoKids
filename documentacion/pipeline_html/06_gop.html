  326  def _distancia_edicion(a, b):
  327      """Distancia de Levenshtein entre dos listas de símbolos -- cuántas
  328      inserciones/borrados/sustituciones hacen falta para convertir `a` en
  329      `b`. Programación dinámica clásica, O(len(a)*len(b))."""
  330      n, m = len(a), len(b)
  331      dp = [[0] * (m + 1) for _ in range(n + 1)]
  332      for i in range(n + 1):
  333          dp[i][0] = i
  334      for j in range(m + 1):
  335          dp[0][j] = j
  336      for i in range(1, n + 1):
  337          for j in range(1, m + 1):
  338              if a[i - 1] == b[j - 1]:
  339                  dp[i][j] = dp[i - 1][j - 1]
  340              else:
  341                  dp[i][j] = 1 + min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1])
  342      return dp[n][m]

  424

  427      # "l" esperada que venga justo después de una consonante de sinfón.

  443      for tipo, esperado, reconocido in ops:

  445          if tabla is None:
  446              continue

  559  def _localizar_subsecuencia(secuencia, objetivo):
  560      """Busca dónde aparece `objetivo` (lista de fonemas de la palabra sola)
  561      dentro de `secuencia` (lista de fonemas de la frase completa), probando
  562      todas las ventanas de largo parecido y quedándose con la de menor
  563      distancia de edición. No exige coincidencia exacta a propósito: la
  564      regla de "r fuerte al principio de palabra" en texto_a_fonemas() mira
  565      el inicio del STRING completo, así que una palabra que empieza con r
  566      en medio de una frase puede fonemizarse levemente distinto que la
  567      misma palabra calculada sola -- por eso se busca la MEJOR ventana, no
  568      un substring exacto.
  569  
  570      Devuelve (inicio, fin) del mejor tramo dentro de `secuencia`."""
  571      n, m = len(secuencia), len(objetivo)
  572      if m == 0 or n == 0:
  573          return 0, 0
  574      mejor_inicio, mejor_fin, mejor_dist = 0, min(m, n), None
  575      for largo in (max(1, m - 1), m, m + 1):
  576          for inicio in range(0, max(1, n - largo + 1)):
  577              fin = inicio + largo
  578              if fin > n:
  579                  continue
  580              dist = _distancia_edicion(secuencia[inicio:fin], objetivo)
  581              if mejor_dist is None or dist < mejor_dist:
  582                  mejor_inicio, mejor_fin, mejor_dist = inicio, fin, dist
