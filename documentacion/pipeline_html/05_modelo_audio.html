  128  def stratified_split(y, val_fraction, seed):
  129      rng = np.random.RandomState(seed)
  130      train_idx, val_idx = [], []
  131      for c in np.unique(y):
  132          idx = np.where(y == c)[0]
  133          rng.shuffle(idx)
  134          n_val = max(1, int(round(len(idx) * val_fraction)))
  135          val_idx.extend(idx[:n_val])
  136          train_idx.extend(idx[n_val:])

  139      return np.array(train_idx), np.array(val_idx)

 1070  def _classify_audio(audio_path, palabra):
 1071      """Clasificador de audio ENTRENADO con clips reales (ver
 1072      train_audio_classifier.py) -- a diferencia del GOP de arriba (que fuerza
 1073      el audio contra la palabra esperada pase lo que pase), esto predice
 1074      directamente entre las clases que existan para esa palabra
 1075      ("<palabra>_correcto", "<palabra>_incorrecto_*"), así que SÍ puede
 1076      reconocer que el audio no se parece a nada de lo esperado.
 1077  
 1078      Devuelve {"clase_predicha", "prob", "es_correcto"} o None si no hay
 1079      clasificador entrenado, o si esa palabra no tiene clases entrenadas
 1080      todavía (el clasificador solo conoce las palabras con clips grabados)."""
 1081      clf = _state["audio_clf"]
 1082      if clf is None or audio_path is None or not palabra:
 1083          return None
 1084      class_names = _state["audio_clf_classes"]
 1085      # Solo tiene sentido clasificar si HAY clases de esta palabra entre las
 1086      # que el clasificador aprendió -- si pedís "perro" pero solo entrenaste
 1087      # con "carro", no hay nada que comparar.
 1088      #
 1089      # BUG real detectado (caso "perro"): un simple startswith(f"{palabra}_")
 1090      # también agarraba las sub-familias de RR ("perro_medias_correcto",
 1091      # "perro_doble_correcto") cuando se pedía la palabra COMPLETA ("perro"),
 1092      # porque esos nombres también empiezan con "perro_". Eso contaminaba la
 1093      # familia: decir "perro" mal podía terminar comparado contra clips de
 1094      # "perro a medias" (un pedacito, acústicamente muy distinto), no contra
 1095      # un "perro" completo mal dicho. Ahora matchea SOLO "<palabra>_correcto"
 1096      # o "<palabra>_incorrecto*" -- las sub-familias piden su propio prefijo
 1097      # exacto (ej. "perro_medias"), así que no se cuelan acá.
 1098      indices_de_la_palabra = [
 1099          i for i, c in enumerate(class_names)
 1100          if c == f"{palabra}_correcto" or c.startswith(f"{palabra}_incorrecto")
 1101      ]
 1102      if not indices_de_la_palabra:
 1103          return None
 1104      try:
 1105          emb = extract_embedding(audio_path, _state["audio_model"], _state["audio_feature_extractor"], DEVICE)
 1106          emb_norm = (emb - _state["audio_clf_mean"]) / _state["audio_clf_std"]
 1107          # mean/std se guardaron con shape (1, 1024) (ver train_audio_
 1108          # classifier.py) -- emb_norm ya sale (1, 1024) por el broadcasting,
 1109          # así que reshape en vez de unsqueeze (que agregaba una dimensión de
 1110          # más y rompía el softmax/indexado de abajo).
 1111          x = torch.from_numpy(emb_norm.reshape(1, -1)).float().to(DEVICE)
 1112          with torch.no_grad():
 1113              probs = torch.softmax(clf(x), dim=1)[0]
 1114          # BUG real detectado: acá se calculaba el argmax sobre las 58 clases
 1115          # completas, sin usar "indices_de_la_palabra" para nada -- si pedías
 1116          # "perro" pero el audio se parecía más a "gota", devolvía
 1117          # "gota_correcto" en vez de reconocer que ni siquiera es la palabra
 1118          # pedida. Ahora el argmax se restringe a los índices de ESA palabra
 1119          # (correcto/incorrecto/medias/doble, lo que tenga entrenado) -- si el
 1120          # audio no se parece a nada de esa familia, la probabilidad ahí
 1121          # dentro sale baja de por sí (la masa está en otras clases), así que
 1122          # cae solo bajo AUDIO_MIN_CONFIDENCE sin necesitar lógica aparte.
 1123          probs_familia = probs[indices_de_la_palabra]
 1124          idx_local = int(torch.argmax(probs_familia).item())
 1125          idx = indices_de_la_palabra[idx_local]
 1126          clase_predicha = class_names[idx]
 1127          return {
 1128              "clase_predicha": clase_predicha,
 1129              "prob": float(probs[idx].item()),
 1130              "es_correcto": clase_predicha == f"{palabra}_correcto",
 1131          }
 1132      except Exception as e:
 1133          print(f"[AVISO] Falló el clasificador de audio entrenado ({e}) -- sigo solo con el GOP.")
 1134          return None
