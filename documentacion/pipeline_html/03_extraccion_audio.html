  246      """Devuelve un tensor 1D float32 mono a 16kHz, sea cual sea el
  247      sample rate/canales originales del .wav grabado."""

  263  def extract_embedding(wav_path, model, feature_extractor, device):
  264      """Vector fijo (EMBEDDING_DIM,) que resume el AUDIO completo del clip --
  265      a diferencia de score_pronunciation() (que da un puntaje por fonema
  266      esperado), esto es lo que usa el clasificador entrenado con datos reales
  267      (train_audio_classifier.py) para decidir directamente "correcto" vs
  268      cada tipo de error, igual que el modelo visual hace con landmarks.
  269  
  270      Es el promedio en el tiempo del último estado oculto del encoder wav2vec2
  271      (antes de la cabeza CTC de fonemas) -- una representación acústica
  272      general del clip, no atada a ninguna palabra esperada."""
  273      waveform = load_audio_16k(wav_path)
  274      inputs = feature_extractor(waveform.numpy(), sampling_rate=TARGET_SAMPLE_RATE, return_tensors="pt")
  275      with torch.inference_mode():
  276          # model.wav2vec2 es el encoder base (sin la cabeza de fonemas) --
  277          # Wav2Vec2ForCTC lo expone como submódulo con ese nombre.
  278          hidden = model.wav2vec2(inputs.input_values.to(device)).last_hidden_state  # (1, T, 1024)
  279          pooled = hidden.mean(dim=1).squeeze(0)  # (1024,)
  280      return pooled.cpu().numpy()
